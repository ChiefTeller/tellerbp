החכם ואלוהים

סָלִיק חַכִּים חַד לְגִנְתָא דְעֵדֶן,
וְתַמָּן פָּגַע בֵּאלָהָא.

אֲמַר לֵיהּ אֱלָהָא:
הַחַכִּים אַתְּ?

וְהָא לָא בָּעֵינָא אֲנָא דְיִקְרוּן לִי חַכִּים,
וְאַתְּ בָּעֵי דָּא?

לָאו חַכִּים הוּא
דְעָבֵיד עֲבִידְתֵיהּ בִּרְחִימוּתָא,
בְּלָא דְתָבַע יְקָר אוֹ בִּרְכְתָא?

תרגום לעברית

חכם אחד עולה לגן עדן, ושם פגש את אלוהים. אמר לו אלוהים: האם חכם אתה? הרי אינני מבקש אני שיקראו לי חכם, אך מבקש זאת אתה? האין חכם מי שעושה את מלאכתו באהבה, בלי לדרוש כבוד או ברכה?

וְהָא כָּל עֲבִידְתָא שָׁוְיָא קֳדָמַי.
לָא חַכִּים הוּא אוּמָנָא
דְעָבֵיד מָאנָא דְחַרְסָא שַׁפִּיר,
וּמִשְׁתַּמְּשִׁין בֵּיהּ בְּנֵי נָשָׁא?

וְלָא חֲמָרָא חַכִּים הוּא,
כַּד טָעֵין עַל גַּבֵּיהּ סְחוֹרְתָא דַאֲחֵרָנִין,
וְלָא בָּעֵי יְקָר אוֹ רְחִימוּתָא?

וְלָא מַיָּא חַכִּימִין אִינוּן,
כַּד מַהְנִין לְכוֹלָּא
וְלָא בָּעוּ דְיִקְרוּן לְהוֹן בַּעֲלֵי חָכְמְתָא?

תרגום לעברית

והרי כל מלאכה שווה בעיני. האין חכם הוא האומן
היוצר כלי חרס יפה, ובני אדם משתמשים בו? ואין החמור חכם, כאשר הוא סוחב על גבו סוחרה של אחרים, ואינו מבקש כבוד או אהבה? ואין המים חכמים, כאשר הם מועילים לכולם ואינם מבקשים שיקראו להם בעלי חוכמה?

וְלָא בַּר נָשׁ חַכִּים הוּא,
כַּד יָדַע דַּאֲנָא רָזָא אֲנָא,
וְהוּא לָא יָדַע מִדָּעַם?

חַכִּים דָּמֵי לְמָאנָא דְמָלֵי תְדִיר,
וְלָא שָׁבֵיק אֲתַר לְמַיָּא.

בְּגִין כָּךְ אֲמִינָא:
חַכִּים הוּא מַאן דְּלָא בָּעֵי
דְיִקְרוּן לֵיהּ חַכִּים.

תרגום לעברית

ואין האדם חכם, כאשר הוא יודע שאני המסתורין,
והוא אינו יודע דבר? חכם דומה לכלי שנשאר מלא תמיד, ואינו משאיר מקום למים. לכן אומר אני:
חכם הוא מי שאינו מבקש שיקראו לו חכם.

הוּא דָּמֵי לְמָאנָא רֵיקָא,
דְּשָׁבֵיק לְחָכְמְתָא לְמֵיעַל לְגַוֵּיהּ.

הוּא דָּמֵי לַחֲמָרָא דְטָרַח,
וְלָא בָּעֵי דְיִקְרוּן לֵיהּ חַכִּים.

וְהוּא דָּמֵי לְמַיָּא
דְיָהְבִין חַיֵּי,
וְלָא תָבְעִין מִדָּעַם.

תרגום לעברית

הוא כמו כלי ריק, המאפשר לחוכמה להיכנס אליו. הוא כמו חמור עמל, שאינו מבקש שיקראו לו חכם. הוא כמו המים, הנותנים חיים, ואינם דורשים דבר.

האיש והמלאך

בַּר נָשׁ אָזֵיל בְּחוּרְשָׁא,
וּפָגַע בְּמַלְאָכָא.

אָמַר גַּבְרָא לְמַלְאָכָא:
עָבֵיד אֲנָא עֲבִידְתָא דֵּאלָהָא מִן עַלְמֵי עָלְמַיָּא,
וּמְיַקַר אֲנָא לְאַבְהָתַי,
וּמְיַקַר אֲנָא לִבְרִיָּתָא,
וּמְיַקַר אֲנָא לִבְנֵי נָשָׁא,
וְלֵית בִּי יֵצֶר בִּישׁ.

תרגום לעברית

אדם הולך ביער, ופוגש במלאך. האיש אומר למלאך:
אני עושה את עבודת אלוהים במשך עידנים, מכבד את ההורים, מכבד את הבריות, מכבד את האנשים
ואין בי היצר הרע.

בְּרַם אֱלָהָא מְרוֹמֵם לַאֲחֵרָנִין,
וְלִי לָא.

הוּא עָבֵיד לַאֲחֵרָנִין יְדִיעִין,
וְלִי לָא.

הוּא יָהֵיב עוּתְרָא לַאֲחֵרָנִין,
וְלִי לָא.

כְּדֵין שְׁאָלֵיהּ מַלְאָכָא:

הֲמִן יוֹמָךְ פְּגַעְתָּ בֵּאלָהָא בִּיקָרֵיהּ וּבִגְרָמֵיהּ?
אוֹ דִּילְמָא חַכִּימָא רַבָּא טָמִיר הוּא?

תרגום לעברית

אבל אלוהים מרומם אחרים, ואותי לא. הוא הופך אחרים למוכרים, ואותי לא. הוא נותן עושר לאחרים,
ולי לא. אז שאל אותו המלאך: האם פעם פגשת את אלוהים בכבודו ובעצמו? או שהחכם הגדול הוא נסתר?

הֲחֲזֵיתַנִי מְהַלֵּךְ בִּלְבוּשִׁין יַקִּירִין בְּבֵי מִדְרָשָׁא?
אוֹ דִּילְמָא חַכִּימָא דְּכַוָּתִי קָאֵים אַלְמוֹנִי בְּחוּרְשָׁא?

הֲתִּקְרֵי לֵאלָהָא אוֹ לִי מִסְכְּנִין,
מִטּוּל דְּלֵית לָן עוּתְרָא סַגִּי?

כְּפִין אַתְּ לְמֵיכְלָא,
אוֹ כְּפִין אַתְּ לְיֵצֶר דְּתַשְׁקוּתָא?

תרגום לעברית

האם ראית אותי הולך בבגדים יקרים בבית המדרש? או שחכם כמותי נשאר אלמוני בחורשה? האם תקרא לאלוהים או לי עניים? כי איננו מחזיקים בעושר רב? האם רעב אתה למאכל? או רעב אתה ליצר התשוקה?

עָנָה גַּבְרָא וַאֲמַר:
כְּפִין אֲנָא לְתַשְׁקוּתָא,
וְלָא לְמֵיכְלָא.

וַאֲמַר לֵיהּ מַלְאָכָא:

אִם בָּעֵי אַתְּ לְמִפְלַח לֵאלָהָךְ,
אִידַּמֵּי לֵיהּ וְהַעֲבֵר תַּשְׁקוּתָא.

אִם בָּעֵי אַתְּ לְמִטַּר צָרְכָךְ,
אֱכוֹל וְתוּ לָא.

תרגום לעברית

האיש ענה: רעב אני לתשוקה ולא למאכל. והמלאך השיב לו: אם אתה רוצה לעבוד את אלוהיך, הידמה לו והסר תשוקה. אם אתה רוצה לשמור על צרכיך, תאכל ותו לא.

זהו רצון אלוה

יְהוּדָאָה חַכִּימָא סָלִיק לִשְׁמַיָּא,
פָּגַע בֵּאלָהָא וַאֲמַר לֵיהּ:

אֱלָהַנָא, מָרֵי עָלְמָא,
בְּרִיךְ אַנְתְּ לְעָלַם.
שַׁלֵּימִית חַיַּי בְּעָלְמָא דְחוּמְרָא מְגֻשָּׁמָא,
וַאֲנָא קָאֵים קֳדָמָךְ כְּצַדִּיק עָלְמָא.

תרגום לעברית

יהודי חכם עולה השמימה

הוא פוגש את אלוהים ואומר לו

אלוהינו, ריבון העולם, מבורך אתה לעולם

סיימתי את חיי בעולם החומרי המגולם

אני ניצב לפניך כצדיק עולם

עֲבַדִית פִּתְגָמָךְ בְּדִקְדּוּקָא יַתִּירָא,
נְטָרִית עָלְמָא דְאוֹרַיְתָא,
קַיֵּימִית פִּקּוּדָךְ בְּלָא פְּשָׁרָא,
וּכְעַן בָּעֵינָא לְמֵיעַל בְּתַרְעָךְ.

תרגום לעברית

עשיתי דבריך בדיקדוק יתרה

שמרתי על עולם התורה

קיימתי את מצוותיך ללא פשרה

וכעת אני מבקש להיכנס בשערך

אֲמַר לֵיהּ אֱלָהָא:

בְּרִי חַכִּימָא, לָא אֲמַר אֲנָא תּוֹדָה,
אֲרֵי בְּרֵאת לָךְ כֹּלָּא,
וְהוּא כְּסִי מִן עֵינָךְ.

בָּעֵית פְּשִׁיטוּתָא,
וְאַנְתְּ לְבַשְׁתְּ גְּלִימָא.
בְּרֵאת שִׁמְשָׁא,
וְאַנְתְּ אִתְעַטַּפְתְּ וְאִטְמַרְתְּ מִשּׁוּפְרָהּ.

יְהַבִית לָךְ רִבּוּיָא וּרְשׁוּ לְמִבְחַר,
וְאַנְתְּ בְּחַרְתְּ לְדַקְדֵּק דַּוְקָא בְּאוֹרַיְתָא.

תרגום לעברית

עונה לו אלוהים

ילדי החכם אינני אומר תודה

כי בראתי לך את הכל אך נסתר מעינך

ביקשתי פשטות אך לבשת גלימה

בראתי את השמש אך התעטפת והסתתרת מיופיה

נתתי לך מגוון וזכות בחירה

אך בחרת לדקדק דווקא את התורה

כַּד בָּעֵית דִּתְהַלֵּךְ בְּאוֹרְחָתַי,
בְּעֵית לָךְ רַבָּא,
וּפָרֵשְׁתְּ מִלִּין עַד אֵימָתַי?

בַּאֲתַר לְמֵידַע פְּשִׁיטוּתָא דְמִלַּיָּא,
בְּעֵית לָךְ בְּבַר נָשׁ אֱלָהִין אָחֳרָנִין.

בָּעֵית מִנָּךְ דְלָא תַעֲבֵיד צְלֵם וְלָא דְמוּתָא,
וְאַנְתְּ עֲבַדְתְּ מִן רַבָּךְ חִלּוּפָא לִבְרִיאָה.

תרגום לעברית

כאשר רציתי שתלך בדרכיי

חיפשת לך רב ופירשת עד מתי?

במקום להבין את פשטות הדברים

חיפשת לך באדם אלוהים אחרים

ביקשתי ממך להימנע מפסל או מתמונה

אך הפכת את רבך לתחליף הבריאה

יְהַבִית לְנִשְׁמָתָא סַגִּיאָתָא
לְמֶעְבַּד בִּיצִירְתָא,
וְאַנְתְּ חֲזֵית בְּהוֹן
רַק עֲגַלְתָּא רֵיקַנְתָּא.

וְאֵיכְדֵין יָכְלָא יְצִירְתִּי עִלָּאָה
לְמֶהֱוֵי עֲגַלְתָּא רֵיקַנְתָּא
אוֹ חוֹבָא רַבָּא?

לְטַעֲמֵיהּ דְרַבָּךְ
אוֹ לְטַעֲמָךְ,
וְכִי אֲנָא הוּא דְחָטֵית,
אוֹ טְעֵית אֲנָא זְעֵיר?

תרגום לעברית

נתתי לנשמות רבות לפעול ביצירה

אך ראית בהם רק עגלה ריקה

וכיצד יכולה להיות יצירתי הנעלה

עגלה ריקה או חטא נורא

האם לטעם רבך או לטעמך

אני החוטא או טעיתי קמעה?

צִמְצַמְתְּ פֵּירוּשָׁא דִגְנֵיבָה,
כַּד אֲנָא פַקְּדִית בְּכַוָּנָא בִּכְלָלָא.

אֲרֵי מַן דְגָנֵיב דַּעְתָּא,
אוֹ שָׁקֵיל מִן קוּפָּה,
בִּשְׁקָלֵיהּ מְחַסֵּר הוּא
מִן חַלָּשָׁא אוֹ מִן אַרְמַלְתָּא.

תרגום לעברית

צמצמת את פירוש הגניבה

כאשר אני ציוויתי בכוונה בהכללה

כי הגונב דעת או לוקח מתוך קופה

בלוקחו מחסיר הוא מחלש או מאלמנה

וּכְעַן אַנְתְּ סָלֵיק וּבָעֵי בִּרְכְתָא,
דְאַעֵיל יָתָךְ בְּתַרְעַי,
בָּתַר דְלָא יְלֵפְתְּ מִידַּעַם.

עֲסַקְתְּ בְּמִלֵּי דַאֲחֵרִין,
וּשְׁבַקְתְּ לֵאלָהָךְ.

תרגום לעברית

וכעת אתה עולה ומבקש ברכה

שאכניס אותך בשעריי אחרי שלא למדת מאומה

עסקת בדברי אחרים וזנחת את אלוהיך

וְכַד יְדַעְתְּ בְּגוֹ לִבָּךְ
דְעוֹבָדֵיהוֹן טָעָן,
וּבִידָךְ רְשׁוּתָא לְמִבְחַר,

בְּחַרְתְּ לְמִשְׁתַּקָּא,
וְלָא אִתְפְּנֵית לְוָתִי בִּבְעוּתָא,
לְמִסְעַד לְמֵיתֵי צֶדֶק,
שַׁוְיוּתָא וְאַחֲוָתָא
לִבְנֵי כָּל בְּרִיאָה.

וַהֲלָא יָכֵיל הֲוֵית
לַאֲקָמָא אַף רַעֲשָׁא,
קֳבֵיל עַוְלָא דְאִתְחֲזֵי
וְלָא אִשְׁתְּמַע.

תרגום לעברית

וכאשר ידעת בתוך ליבך

כי מעשיהם טועים ובידך הבחירה

בחרת לשתוק ולא פנית אלי בבקשה

לעזור להביא צדק שוויון ואחווה לבני כל הבריאה

הרי יכולת להקים גם מהומה

אל מול עוול שנראה ואינו נשמע

בְּרַם בְּחַרְתְּ לְצַמְצְמָא חֶזְוָךְ,
וְיַת אוֹרַיְתִי דִּקְשׁוֹט
שְׁכַחְתְּ בִּמְהֵירָא.

מִטּוּל הָכֵין
לָא תֵיעוֹל בְּתַרְעַי.

תִּשְׁתַּלַּח לְאַרְעָא תּוּב -
וְעַד אֵימַת?

עַד דְתֵיקוּם בִּגְבוּרְתָא
בְּכָל קְיָמַי,
דִּכְתִיבִין בְּלִבָּךְ
וְלָא בִּלְשָׁן אֲרָמָיָא.

תרגום לעברית

אך בחרת לצמצם ראייתך

ואת תורתי האמיתית שכחת במהרה

על כן לא תכנס בשעריי

אתה תשלח לארץ שוב ועד מתי

שתעמוד בגבורה בכל חוקותי

הכתובים בלבבך ולא בארמיי
